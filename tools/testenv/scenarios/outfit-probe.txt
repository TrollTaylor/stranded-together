# variation removed, our exos variation added, our suit colour set), then explicit outfits.
#   pwsh -File tools\testenv\Start-TestRun.ps1 -Scenario outfit-probe -SeedSaves complete-100
# covers: outfit

waitlog 120 \[overlay\] input attached
