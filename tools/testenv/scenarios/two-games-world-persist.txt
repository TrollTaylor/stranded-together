# Shared structures W1 (docs/world-sync.md): persistence in the host's save, hidden, on the launcher's own Local test files:
#   pwsh -File tools\testenv\Start-TestRun.ps1 -Scenario two-games-world-persist -SeedSaves complete-100 -GuestSeed profile-only
# The host builds a watchtower, the guest builds a timefall shelter (held, created by the host, mirrored back); a save of
# the host's game is written (save_guard 1 4: the game's own autosave, into the run's scratch copy) and loaded in the world
# (save_guard 3; the guest follows: it loads the host's new save and joins again). Pass: after the load each world has each
# structure exactly once (no duplicate, nothing lost), offline 1 on the host, the same ids; the guest's second join check
# binds them (created 0: they came from the host's save, not from the mod), no FAULT, no OTGC WRITE, both games exit 0.
role host
graphics low
timeout 360
launcherini local-test
start control 200
waitlog 10 \[world\] world sync ON \(host\)
@guest waitlog 120 \[world\] guest: snapshot #1 complete: bound [0-9]+, created 0, removed 0, replaced 0
@both goto mknot-road
wait 5000
action dev_build 9 0 12
waitnew 5 \[world\] dev build: id [0-9]+ type 9 \(watchtower\) anim 0
@guest waitlog 10 \[world\] guest: mirror hid [0-9]+ -> gid [0-9]+ \(exact\) type 9 \(watchtower\) anim 0
@guest action dev_build 18 0 20
@guest waitlog 10 \[world\] guest: local build type 18 \(timefall shelter\) held, request #1 sent
waitlog 10 \[world\] host: the guest's build #1: created hid [0-9]+ type 18 \(timefall shelter\) offline 1
@guest waitlog 10 \[world\] guest: build #1 confirmed as hid [0-9]+
wait 6000
action world_sync 10 -1 60
waitnew 3 \[world\] check: type -1 within 60 m: 2 \(ids [0-9]+ [0-9]+\) offline 2
@guest action world_sync 10 -1 60
@guest waitnew 3 \[world\] check: type -1 within 60 m: 2 \(ids [0-9]+ [0-9]+\) offline 2
screenshot P0-host-before-save
@guest screenshot P0-guest-before-save
# ---- the host's game saves (its own autosave, into the run's scratch copy), then loads that save in the world
until 30 1500 \[save\] #[0-9]+ pending \(forced save F[0-9]+\): auto save => action save_guard 1 4 1
waitlog 60 \[save\] finished ok: #[0-9]+ \(forced save F[0-9]+\)
wait 2000
action save_guard 3
waitnew 20 \[coop\] host: in-world Load \(type 0x4
@guest waitnew 120 \[coop\] guest: wrote the host's save #[0-9]+ .*
@guest waitnew 180 \[coop\] guest: in the world after .* save #[0-9]+
waitnew 120 \[coop\] host: the partner's RESULT for load #[0-9]+: ok
@guest waitlog 60 \[world\] guest: snapshot #[2-9] complete: bound [0-9]+, created 0, removed 0, replaced 0
waitlog 10 \[world\] host: world ready \(world [2-9]\): [0-9]+ shared structures
wait 4000
action world_sync 10 -1 60
waitnew 3 \[world\] check: type -1 within 60 m: 2 \(ids [0-9]+ [0-9]+\) offline 2
@guest action world_sync 10 -1 60
@guest waitnew 3 \[world\] check: type -1 within 60 m: 2 \(ids [0-9]+ [0-9]+\) offline 2
action world_sync 10 9 60
@guest action world_sync 10 9 60
action world_sync 10 18 60
@guest action world_sync 10 18 60
action world_dump -1 6
@guest action world_dump -1 6
screenshot P1-host-after-load
@guest screenshot P1-guest-after-load
@both action world_sync 3
nolog \[world\].*FAULT
@guest nolog \[world\].*FAULT
nolog \[gate\] OTGC WRITE
@guest nolog \[gate\] OTGC WRITE
nolog \[world\] guest: mirror FAIL
@guest nolog \[world\] guest: mirror FAIL
@guest quit
wait 3000
quit
