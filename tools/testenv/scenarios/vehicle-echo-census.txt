# Vehicles, research (docs/real-partner-body.md section 22): the census of the 100% save's vehicles, their looks (paint:
# variations, shader variables, their resources' gold / silver / broken / ride-type variations), the cargo in their
# baggage carriers and their child entities (action vehicle_sync 19), then the truck and the trike ridden with the echo
# partner (its copy 4 m beside ours) for screenshots of our vehicle against its copy.
#   pwsh -File tools\testenv\Start-TestRun.ps1 -Scenario vehicle-echo-census -SeedSaves complete-100
role echo
timeout 300
ini partner_body=real
ini echo_offset_m=4
ini echo_delay_s=0.5
ini partner_collision=0
ini vehicle_sync=1
waitready 90
waitlog 120 \[overlay\] input attached
wait 25000
pad A 150
wait 4000
until 240 5000 \[sync\] local player entity 0+ -> 0*[1-9A-F] => pad A 120
waitplayer 60
until 120 500 ly=32760 -> ok: .*player moved ([1-9]|[1-9][0-9]+)\.[0-9]+ m => pad none 1000 ly=32760
waitlog 40 \[sync\] partner body: real
wait 1500
action vehicle_sync 7
action vehicle_sync 19 0
# ---- the truck: ours and the echo's copy
action vehicle_sync 8 2.2 2
wait 2500
screenshot 01-at-truck
until 20 4000 \[vehicle\] ours: truck => pad X 150
?waitlog 10 \[vehicle\] copy 'truck' (shown|NOT shown)
wait 3500
screenshot 02-truck-and-copy
pad none 700 rx=32767
wait 400
screenshot 02-truck-and-copy-b
action vehicle_sync 19 2
pad X 150
wait 4000
action vehicle_sync 0
# ---- the trike
action vehicle_sync 8 1.4 3
wait 2500
screenshot 03-at-trike
until 20 4000 \[vehicle\] ours: trike (parked|getting|riding) => pad X 150
?waitlog 10 \[vehicle\] copy 'trike' (shown|NOT shown)
wait 3500
screenshot 04-trike-and-copy
pad none 700 rx=32767
wait 400

action vehicle_sync 19 3
pad X 150
wait 3500
action vehicle_sync 0
