# Partner sounds, Phase 3, one game (docs/audio-research.md section 16): role echo, sound_sync=1 with this module's own
# hooks (no audio probe).
#   pwsh -File tools\testenv\Start-TestRun.ps1 -Scenario sound-sync-echo -SeedSaves complete-100
# The echo partner (a real Sam body replaying our own movement 1.5 m to the side, 2 s later) gets our own footsteps,
# foley and gun sounds back through STA1 encode / decode: posted on our Wwise objects at the echo body's joints, at the
# pose play time. Every mirrored post is logged on both sides (`sound_sync 1 1`). Then the cuff map is opened right
# after a run: the items still due are not posted while it is open (and nothing is sent while it is open); the map shows a "new ranked event" notice first: A takes it away, then B closes the map.
# covers: sound
role echo
timeout 360
ini partner_body=real
ini partner_character=sam
ini echo_delay_s=2.0
ini echo_offset_m=1.5
ini sound_sync=1
waitready 90
waitlog 120 \[overlay\] input attached
wait 25000
pad A 150
wait 4000
until 240 5000 \[sync\] local player entity 0+ -> 0*[1-9A-F] => pad A 120
waitplayer 60
until 120 500 ly=32760 -> ok: .*player moved ([1-9]|[1-9][0-9]+)\.[0-9]+ m => pad none 1000 ly=32760
waitlog 40 \[sync\] partner body: real
wait 3000
action sound_sync 1 1
screenshot 01-start
# walk and run, there and back
log sound: walk and run
pad none 4000 ly=16000
pad none 4000 ly=-16000
pad none 4000 ly=32767
pad none 4000 ly=-32767
pad none 0
wait 3000
waitlog 10 \[psound\] mirror post steps
waitlog 10 \[psound\] partner post steps ev 0x[0-9A-F]+ \(their pid [0-9]+\) -> pid [1-9][0-9]*
action sound_sync 0
# jumps
log sound: jump
pad none 1200 ly=32767 keep
pad A 200 ly=32767 keep
pad none 900 ly=32767 keep
pad A 200 ly=32767 keep
pad none 1500 ly=-32767 keep
pad none 0
wait 3000
# a gun: the equipment wheel's upper left item (the assault rifle in the 100% save), aimed and fired
log sound: gun
pad dpad_right 600 keep
pad dpad_right 900 lx=-23170 ly=23170 rx=-23170 ry=23170 keep
pad dpad_right 400 keep
pad none 0
wait 3000
pad none 1500 lt=255 keep
pad none 400 lt=255 rt=255 keep
pad none 300 lt=255 keep
pad none 400 lt=255 rt=255 keep
pad none 0
wait 3000
screenshot 02-gun
pad dpad_right 600 keep
pad dpad_right 900 lx=32767 ly=0 rx=32767 ry=0 keep
pad dpad_right 400 keep
pad none 0
wait 3000
action sound_sync 0
# run, then the cuff map at once: the echo's items still due (2 s) arrive while it is open
log sound: menu
pad none 3000 ly=32767
pad START 150
waitlog 5 \[psound\] gate closed: paused / menu / cutscene \(HandcuffDevice\)
log sound: menu open
wait 4000
screenshot 03-menu
action sound_sync 0
log sound: menu closing
pad A 150
wait 2000
until 20 2500 \[psound\] gate open again .*HandcuffDevice => pad B 150
log sound: menu closed
action sound_sync 0
# idle: whatever is still playing ends (end-of-event callbacks)
wait 5000
log sound: end
action sound_sync 1 0
action sound_sync 0
action log_status
quit
