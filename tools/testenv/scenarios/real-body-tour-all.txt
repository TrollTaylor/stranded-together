# Real partner body, every body_table id (docs/real-partner-body.md section 2, "offered"):
#   pwsh -File tools\testenv\Start-TestRun.ps1 -Scenario real-body-tour-all -SeedSaves complete-100
# Spawns ids 1..16 in turn, each 5 m ahead of Sam and mirroring his live pose. Frames per body:
#   NN-<key>-idle      the body stands in Sam's idle, facing the camera;
#   NN-<key>-walk-a/b  Sam turns round and walks towards the camera (the pad stays held); the body, now
#                      behind him, walks in place and is seen past his shoulder.
# Sam then walks away from the camera again. These timings (0.7 s + 0.2 s towards, 1.1 s away) kept all 16
# bodies framed alike, standing on open floor (run 20261116-194441). Longer walks drifted Sam until the
# body stood behind a parked vehicle (run 20261116-195757), and walking sideways (194938) or towards the
# body (195431) turned the camera, so Sam ended up facing walls with the body out of frame.
# For each body the log has the load time ("loaded after"), joint count and layout hash ("model ... joints
# 77 layout"), the joint-name map ("mapped by name: N missing") and the pelvis latch ("pelvis ... ratio");
# real_stats adds the hook counters. The supervisor's waitlog searches the whole log, so each body's wait
# names that body only (an earlier body's lines must not end it); it is optional (?), so a body that never
# loads does not end the tour.
role solo
timeout 360
ini pose_hook=1
waitready 90
waitlog 120 \[overlay\] input attached
wait 25000
pad A 150
wait 4000
until 240 5000 \[sync\] local player entity 0+ -> 0*[1-9A-F] => pad A 120
waitplayer 60
until 120 500 ly=32760 -> ok: .*player moved ([1-9]|[1-9][0-9]+)\.[0-9]+ m => pad none 1000 ly=32760
wait 1500
screenshot 00-control
action real_probe
# id 1: sam
action real_spawn 1 5 1
?waitlog 30 \[real\] (.*'sam'( is live| refused|.* did not load| will not be tried)|the game cancelled loading 'sam')
wait 1500
screenshot 01-sam-idle
action real_stats
pad none 700 ly=-32767 keep
screenshot 01-sam-walk-a
wait 200
screenshot 01-sam-walk-b
pad none 0
action real_stats
action real_remove
pad none 1100 ly=32767
wait 600
