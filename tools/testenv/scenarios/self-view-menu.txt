# The cargo menu's Sam figure under the self view (docs/real-partner-body.md section 15).
#   pwsh -File tools\testenv\Start-TestRun.ps1 -Scenario self-view-menu -SeedSaves complete-100
# A guest-like game (role solo, my_character=fragile, self_character=1). The cuff map (START) opens
# with the "Sam" entry focused; A on it is "Manage Cargo", the on-foot cargo menu with Sam's figure
# carrying the backpack and cargo. First with the feature off (`self_view 3 0`: the menu shows Sam),
# switched on while the menu is open (Fragile in the figure's place), the figure turned; closed (B);
# then three more open / close cycles with it on; then the self view off and the menu once more: the
# figure must be Sam again with all his parts (nothing left hidden). B leaves the cargo menu for the
# cuff map, START closes the map. `[menu] subspace state ... (opens N)` counts the menu's openings; `show N` the self view's showings in the field.
# covers: self-view menu-figure
role solo
timeout 360
ini partner_body=real
ini my_character=fragile
ini self_character=1
waitready 90
waitlog 120 \[overlay\] input attached
wait 25000
pad A 150
wait 4000
until 240 5000 \[sync\] local player entity 0+ -> 0*[1-9A-F] => pad A 120
waitplayer 60
until 120 500 ly=32760 -> ok: .*player moved ([1-9]|[1-9][0-9]+)\.[0-9]+ m => pad none 1000 ly=32760
waitlog 40 \[self\] showing 'fragile'
wait 2000
screenshot 01-field-self
# 1: the menu with the feature off (Sam), then on while it is open
action self_view 3 0
pad START 150
wait 3000
screenshot 02-map
until 25 3000 \[menu\] subspace state 5 .*\(opens 1\) => pad A 150
wait 2500
screenshot 03-menu-sam
action self_view 4
action self_view 3 1
waitlog 10 \[menu\] showing 'fragile' in place of the menu's Sam figure
wait 2000
screenshot 04-menu-fragile
action self_view 4
pad none 900 rx=32767
wait 1500
screenshot 05-menu-turned
pad none 900 rx=-32767
wait 1500
screenshot 05b-menu-turned-back
until 25 2500 \[menu\] subspace state 2 .*\(opens 1\) => pad B 150
waitlog 10 \[self\] showing 'fragile'.*show 2\)
wait 1500
screenshot 06-after-close
# B leaves the cargo menu for the cuff map; START closes the map
pad START 150
wait 2500
screenshot 07-field-again
# 2..4: open and close three times with the feature on
pad START 150
wait 3000
until 25 3000 \[menu\] subspace state 5 .*\(opens 2\) => pad A 150
waitlog 10 \[menu\] showing 'fragile' .*show 2\)
wait 1500
screenshot 08-menu-2
until 25 2500 \[menu\] subspace state 2 .*\(opens 2\) => pad B 150
waitlog 10 \[self\] showing 'fragile'.*show 3\)
wait 1000
pad START 150
wait 2500
pad START 150
wait 3000
until 25 3000 \[menu\] subspace state 5 .*\(opens 3\) => pad A 150
waitlog 10 \[menu\] showing 'fragile' .*show 3\)
wait 1500
screenshot 09-menu-3
until 25 2500 \[menu\] subspace state 2 .*\(opens 3\) => pad B 150
waitlog 10 \[self\] showing 'fragile'.*show 4\)
wait 1000
pad START 150
wait 2500
pad START 150
wait 3000
until 25 3000 \[menu\] subspace state 5 .*\(opens 4\) => pad A 150
waitlog 10 \[menu\] showing 'fragile' .*show 4\)
wait 1500
screenshot 10-menu-4
until 25 2500 \[menu\] subspace state 2 .*\(opens 4\) => pad B 150
waitlog 10 \[self\] showing 'fragile'.*show 5\)
wait 1000
pad START 150
wait 2500
screenshot 11-field-after-cycles
pad none 1300 rx=32767
wait 1500
screenshot 12-field-front
# the self view off: Sam in the field, then the menu once more (its figure must be Sam, every part back)
action self_view 0
wait 2000
screenshot 13-field-sam
pad START 150
wait 3000
until 25 3000 \[menu\] subspace state 5 .*\(opens 5\) => pad A 150
wait 2500
screenshot 14-menu-sam-again
action self_view 4
until 25 2500 \[menu\] subspace state 2 .*\(opens 5\) => pad B 150
wait 1500
pad START 150
wait 2500
screenshot 15-field-end
action real_stats
action log_status
quit
